## 4. 数据结构设计

### 4.1 ER 关系图

```

                    │ staff_id ─────────┐              │

                                        │ 1:0..1（跨模块，非物理 FK）
                                        ▼

                                              │ 1:N
                        ┌─────────────────────┼─────────────────────┐
                        │                     │                     │

        ┌──────────────────────┐  ┌──────────────────────┐  ┌──────────────────────┐
        │ sys_password_history │  │ sys_account_log      │  │ sys_account_alias    │
        │ （密码历史）          │  │ （变更日志）          │  │ （登录名别名）        │
        └──────────────────────┘  └──────────────────────┘  └──────────────────────┘
                                                              │
                                ┌─────────────────────────────┘
                                │ 1:N（账号变更时旧名进入别名表）
                                ▼

┌──────────────────────────────┐  1:N  ┌──────────────────────────────────┐
│ sys_account_import_task      │──────►│ sys_account_import_detail        │
│ （导入任务）                  │       │ （导入明细）                      │
└──────────────────────────────┘       └──────────────────────────────────┘

└──────────────────────────────┘
```

**说明：**

- `sys_account.staff_id` 指向 M01 的 `mdm_staff`，**不建物理外键**（跨模块解耦）
- `sys_password_policy` 为单行配置表（`id=1`）
- `sys_account_import_task` / `_detail` 服务于批量导入
- `sys_login_attempt` 记录每次登录尝试，供风控与审计使用

**约束说明**

```sql
-- 一职工一账号：仅对 staff_id 非空生效（部分唯一索引）
CREATE UNIQUE INDEX uk_sys_account_staff
    ON sys_account (staff_id)
    WHERE staff_id IS NOT NULL AND is_deleted = FALSE;

-- 登录名唯一（大小写已在应用层转小写，此处直接唯一）
CREATE UNIQUE INDEX uk_sys_account_username
    ON sys_account (username);
```

**字段说明补充：**

- `username` 存储时统一转小写，`username_display` 保留原始大小写用于展示
- `password_hash` 使用 bcrypt（cost=12），**不可逆**
- `mfa_totp_secret` 使用 AES-256 加密后存储，密钥在配置中心
- `staff_id` 对 `SERVICE` 类型可为空
- `expire_date` 对 `is_long_term=true` 的账号为空

**分区：** 不分区（账号总量 <10,000）

### 4.10 SQLAlchemy 2.0 模型定义

```python
# app/models/sys/account.py
"""账号管理 ORM 模型"""
from datetime import date, datetime
from typing import Optional, List

from sqlalchemy import (
    BigInteger, String, Integer, SmallInteger, Boolean, Date, DateTime,
    Text, JSONB, Index, UniqueConstraint, func,

        UniqueConstraint("username", name="uk_sys_account_username"),

        Index("idx_sys_account_type", "account_type", "status"),

    )

        "PENDING": "待激活", "ACTIVE": "正常", "LOCKED": "已锁定",
        "DISABLED": "已停用", "TERMINATED": "已注销",
    }
    TYPE_LABELS = {
        "EMPLOYEE": "正式职工", "TEMP_EMPLOYEE": "临时职工",
        "EXTERNAL": "外部人员", "SERVICE": "系统集成", "SUPER_ADMIN": "超级管理员",
    }

    def status_label(self) -> str:
        return self.STATUS_LABELS.get(self.status, self.status)

    @property
    def type_label(self) -> str:
        return self.TYPE_LABELS.get(self.account_type, self.account_type)

    def is_active(self) -> bool:
        """是否可用（正常且未过期）"""
        if self.status != "ACTIVE":
            return False
        if self.expire_date and self.expire_date < date.today():
            return False
        return True

    def is_locked(self, now: Optional[datetime] = None) -> bool:
        """是否处于锁定期"""
        if self.status != "LOCKED":
            return False
        if not self.locked_until:
            return True
        return self.locked_until > (now or datetime.now())

    def can_login(self) -> bool:
        """能否登录（不含密码校验）"""
        return self.status in ("PENDING", "ACTIVE") and self.is_active() or \
               (self.status == "PENDING" and
                (not self.expire_date or self.expire_date >= date.today()))
```

```python
class AccountCreate(BaseModel):
    staff_id: Optional[int] = None
    staff_name: Optional[str] = Field(None, max_length=64)
    username: Optional[str] = Field(None, max_length=32)
    account_type: ACCOUNT_TYPE = "EMPLOYEE"
    owner_dept_id: Optional[int] = None

    init_password_mode: Literal["RANDOM", "MANUAL"] = "RANDOM"
    manual_password: Optional[str] = Field(None, max_length=64)
    force_change_on_first_login: bool = True
    send_password_by_sms: bool = False

    is_long_term: bool = True
    expire_date: Optional[date] = None

    phone: Optional[str] = Field(None, max_length=20)
    phone_backup: Optional[str] = Field(None, max_length=20)

    position_id: Optional[int] = None
    role_ids: Optional[List[int]] = None

    remark: Optional[str] = Field(None, max_length=512)

    @field_validator("username")
    @classmethod
    def validate_username(cls, v):
        if v is None:
            return v
        if not re.match(r"^[A-Za-z][A-Za-z0-9_]{3,31}$", v):
            raise ValueError("登录名格式不正确（4~32 位，首字符字母，可含字母数字下划线）")
        return v.lower()

    @field_validator("phone", "phone_backup")
    @classmethod
    def validate_phone(cls, v):
        if v is None:
            return v
        if not re.match(r"^1[3-9]\d{9}$", v):
            raise ValueError("手机号格式不正确")
        return v

    @model_validator(mode="after")
    def validate_create(self):
        # 职工必填（非外部/服务账号）
        if self.account_type not in ("EXTERNAL", "SERVICE") and not self.staff_id:
            raise ValueError("必须选择关联职工")
        # 手动密码必填且需校验
        if self.init_password_mode == "MANUAL" and not self.manual_password:
            raise ValueError("手动设置模式下必须填写密码")
        # 长期有效则不应有到期日
        if self.is_long_term and self.expire_date:
            raise ValueError("长期有效账号不应设置到期日")
        # 短期账号必须填到期日
        if not self.is_long_term and not self.expire_date:
            raise ValueError("非长期账号必须设置到期日")
        # 到期日不早于今天
        if self.expire_date and self.expire_date < date.today():
            raise ValueError("到期日不能早于今天")
        # 外部账号必须指定负责科室
        if self.account_type == "EXTERNAL" and not self.owner_dept_id:
            raise ValueError("外部人员账号必须指定负责科室")
        # 短信通知必须有手机号
        if self.send_password_by_sms and not self.phone:
            raise ValueError("短信通知必须先填写手机号")
        return self

        if self.new_password != self.confirm_password:
            raise ValueError("两次输入的密码不一致")
        if self.new_password == self.old_password:
            raise ValueError("新密码不能与当前密码相同")

class PasswordPolicyBase(BaseModel):
    min_length: int = Field(8, ge=6, le=32)
    max_length: int = Field(32, ge=8, le=64)
    require_upper: bool = True
    require_lower: bool = True
    require_digit: bool = True
    require_special: bool = False
    forbid_username: bool = True
    forbid_weak: bool = True
    history_count: int = Field(5, ge=0, le=20)
    expire_days: int = Field(90, ge=0, le=365)
    warn_days: int = Field(7, ge=0, le=30)
    max_fail_count: int = Field(5, ge=3, le=10)
    lock_minutes: int = Field(30, ge=1, le=1440)
    first_login_change: bool = True
    exempt_types: Optional[str] = "SERVICE,SUPER_ADMIN"

    @model_validator(mode="after")
    def validate_policy(self):
        if self.min_length > self.max_length:
            raise ValueError("最小长度不能大于最大长度")
        if not any([self.require_upper, self.require_lower,
                    self.require_digit, self.require_special]):
            raise ValueError("至少需要启用一种字符类型要求")
        if self.expire_days > 0 and self.warn_days >= self.expire_days:
            raise ValueError("提醒天数必须小于有效期")
        return self
```
