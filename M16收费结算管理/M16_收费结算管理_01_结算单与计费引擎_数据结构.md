### 4.1 ER 关系图


**设计说明：**

- `m16_settlement` 是结算单**主表**，四种来源（医嘱/挂号/视光/住院）共用
- `m16_settlement_item` 存每一行明细，通过 `order_id`/`order_no` 指回 M07 医嘱
- `m16_price_snapshot` **一行明细对应一条或多条快照**（重算后旧快照保留）
- `m16_settlement_log` 存全量操作轨迹，**不可修改、不可删除**
- `m16_rewrite_queue` 是分布式事务的补偿表，支付成功但回写失败时写入
- **金额字段全部 `NUMERIC`，禁止 `FLOAT`**：浮点数会在累加时产生分位误差

**约束说明**

```sql
ALTER TABLE m16_settlement
    ADD CONSTRAINT ck_m16_settle_no
    CHECK (settlement_no ~ '^BL[0-9]{8}[0-9]{4}$');

ALTER TABLE m16_settlement
    ADD CONSTRAINT ck_m16_settle_source
    CHECK (source_type IN ('ENCOUNTER', 'REGISTRATION',
                            'OPTICAL', 'INPATIENT'));

ALTER TABLE m16_settlement
    ADD CONSTRAINT ck_m16_settle_status
    CHECK (status IN ('UNPAID', 'PAID', 'PART_REFUNDED',
                       'REFUNDED', 'CANCELLED'));

ALTER TABLE m16_settlement
    ADD CONSTRAINT ck_m16_settle_merge
    CHECK (merge_strategy IN ('BY_NONE', 'BY_GROUP', 'BY_DEPT'));

-- 金额非负
ALTER TABLE m16_settlement
    ADD CONSTRAINT ck_m16_settle_amount
    CHECK (total_amount >= 0
       AND discount_amount >= 0
       AND payable_amount >= 0
       AND paid_amount >= 0
       AND refunded_amount >= 0);

-- 应收 = 原价 - 优惠（数值一致性最重要的约束）
ALTER TABLE m16_settlement
    ADD CONSTRAINT ck_m16_settle_payable
    CHECK (payable_amount = total_amount - discount_amount);

-- 优惠不得超过原价
ALTER TABLE m16_settlement
    ADD CONSTRAINT ck_m16_settle_discount
    CHECK (discount_amount <= total_amount);

-- 实收不得超过应收
ALTER TABLE m16_settlement
    ADD CONSTRAINT ck_m16_settle_paid
    CHECK (paid_amount <= payable_amount);

-- 退费不得超过实收
ALTER TABLE m16_settlement
    ADD CONSTRAINT ck_m16_settle_refund
    CHECK (refunded_amount <= paid_amount);

-- 已支付必须有支付时间与方式
ALTER TABLE m16_settlement
    ADD CONSTRAINT ck_m16_settle_paid_required
    CHECK (status NOT IN ('PAID', 'PART_REFUNDED', 'REFUNDED')
           OR (paid_at IS NOT NULL AND paid_method IS NOT NULL));

-- 退费状态必须有退费时间
ALTER TABLE m16_settlement
    ADD CONSTRAINT ck_m16_settle_refunded_required
    CHECK (status NOT IN ('PART_REFUNDED', 'REFUNDED')
           OR (refunded_at IS NOT NULL AND refunded_amount > 0));

-- 全额退费必须退足
ALTER TABLE m16_settlement
    ADD CONSTRAINT ck_m16_settle_refunded_full
    CHECK (status <> 'REFUNDED' OR refunded_amount = paid_amount);

-- 已作废必须有原因
ALTER TABLE m16_settlement
    ADD CONSTRAINT ck_m16_settle_cancelled
    CHECK (status <> 'CANCELLED' OR cancel_reason IS NOT NULL);

-- 来源必填对应 ID
ALTER TABLE m16_settlement
    ADD CONSTRAINT ck_m16_settle_source_ref
    CHECK (
        (source_type = 'ENCOUNTER'    AND encounter_id IS NOT NULL)
     OR (source_type = 'REGISTRATION' AND registration_id IS NOT NULL)
     OR (source_type = 'OPTICAL'      AND optical_order_id IS NOT NULL)
     OR (source_type = 'INPATIENT'    AND inpatient_id IS NOT NULL)
    );

-- 幂等键唯一（部分唯一索引，不含已作废的）
CREATE UNIQUE INDEX uk_m16_settle_idem
    ON m16_settlement (source_type, idempotency_key)
    WHERE idempotency_key IS NOT NULL
      AND status <> 'CANCELLED'
      AND is_deleted = FALSE;

-- 待收费清单查询
CREATE INDEX idx_m16_settle_pending
    ON m16_settlement (settle_date, dept_id, created_at)
    WHERE status = 'UNPAID' AND is_deleted = FALSE;

-- 超时作废扫描
CREATE INDEX idx_m16_settle_expire
    ON m16_settlement (expire_at)
    WHERE status = 'UNPAID' AND is_deleted = FALSE;
```

```sql
-- 数量倍数一致性（核心校验）
ALTER TABLE m16_settlement_item
    ADD CONSTRAINT ck_m16_settle_item_bill_qty
    CHECK (bill_quantity = quantity * eye_multiplier);

-- 眼别与倍数一致性
ALTER TABLE m16_settlement_item
    ADD CONSTRAINT ck_m16_settle_item_eye_mult
    CHECK ((eye_side = 'OU' AND eye_multiplier = 2)
        OR (eye_side <> 'OU' AND eye_multiplier = 1));

-- 金额非负
ALTER TABLE m16_settlement_item
    ADD CONSTRAINT ck_m16_settle_item_amount
    CHECK (quantity > 0
       AND unit_price >= 0
       AND line_amount >= 0
       AND discount_amount >= 0
       AND payable_amount >= 0
       AND refunded_amount >= 0);

-- 行应收 = 行原价 - 行优惠
ALTER TABLE m16_settlement_item
    ADD CONSTRAINT ck_m16_settle_item_payable
    CHECK (payable_amount = line_amount - discount_amount);

-- 优惠不得超过行原价
ALTER TABLE m16_settlement_item
    ADD CONSTRAINT ck_m16_settle_item_discount
    CHECK (discount_amount <= line_amount);

-- 退费不得超过行应收
ALTER TABLE m16_settlement_item
    ADD CONSTRAINT ck_m16_settle_item_refund
    CHECK (refunded_amount <= payable_amount);

-- 来源为医嘱必须有 order_id
ALTER TABLE m16_settlement_item
    ADD CONSTRAINT ck_m16_settle_item_order_ref
    CHECK (source_type <> 'ORDER' OR order_id IS NOT NULL);

-- 已整行退费标记必须一致
ALTER TABLE m16_settlement_item
    ADD CONSTRAINT ck_m16_settle_item_refunded_flag
    CHECK (is_refunded = FALSE OR refunded_amount = payable_amount);

-- 按医嘱反查结算单
CREATE INDEX idx_m16_settle_item_order
    ON m16_settlement_item (order_id)
    WHERE order_id IS NOT NULL AND is_deleted = FALSE;

-- 可退明细
CREATE INDEX idx_m16_settle_item_refundable
    ON m16_settlement_item (settlement_id)
    WHERE refunded_amount < payable_amount AND is_deleted = FALSE;
```
