## 4. 数据结构设计

### 4.1 ER 关系图

```

│ 上游（引用，不建物理外键）                                             │

│  mpi_patient（M03-01，过敏史）                                       │
└──────────────────────────────┬───────────────────────────────────────┘
                               │
                               ▼
┌──────────────────────────────────┐
│ m07_prescription（处方主表）★核心  │
│  · prescription_no / type        │
│  · status（DRAFT→SIGNED→REVIEWED）│
│  · 签名信息（signer/hash/sn）      │

┌──────────────────────────┐  1:N  ┌──────────────────────────┐
│m07_prescription_template │──────►│m07_prescription_template │
│（处方模板）                │       │_item（模板明细）           │
└──────────────────────────┘       └──────────────────────────┘
```

**设计说明：**

- `m07_prescription` 与 `m07_order`（药品类）**一一对应但双向引用**：医嘱存"开单与费用"，处方存"用法与合规"
- `m07_prescription_item.order_id` 关联 M07-03 的医嘱 ID，保证两表一致
- **`m07_medication_check` 独立成表**：合理用药审核需要留痕（谁审的、审出什么、医生如何处置）
- **`m07_antibiotic_consult` 独立成表**：特殊级抗菌药物会诊是独立流程，有审批状态
- 处方**不可物理删除**，只能作废
- **签名后不可修改**：`signature_hash` 固定，任何修改都需作废重开

**约束说明**

```sql
ALTER TABLE m07_prescription
    ADD CONSTRAINT ck_m07_rx_no
    CHECK (prescription_no ~ '^P[0-9]{8}[A-Z0-9]{3}$');

ALTER TABLE m07_prescription
    ADD CONSTRAINT ck_m07_rx_type
    CHECK (prescription_type IN ('NORMAL', 'PEDIATRIC',
                                  'EMERGENCY', 'NARCOTIC'));

ALTER TABLE m07_prescription
    ADD CONSTRAINT ck_m07_rx_paper
    CHECK (paper_color IN ('WHITE', 'GREEN', 'YELLOW', 'RED'));

ALTER TABLE m07_prescription
    ADD CONSTRAINT ck_m07_rx_status
    CHECK (status IN ('DRAFT', 'SIGNED', 'REVIEWED',
                       'RETURNED', 'DISPENSED', 'CANCELLED'));

ALTER TABLE m07_prescription
    ADD CONSTRAINT ck_m07_rx_level
    CHECK (antibiotic_level IS NULL
           OR antibiotic_level IN ('NON_RESTRICTED',
                                    'RESTRICTED', 'SPECIAL'));

ALTER TABLE m07_prescription
    ADD CONSTRAINT ck_m07_rx_sign_method
    CHECK (sign_method IS NULL
           OR sign_method IN ('CA', 'HANDWRITE'));

ALTER TABLE m07_prescription
    ADD CONSTRAINT ck_m07_rx_count CHECK (item_count >= 0);

ALTER TABLE m07_prescription
    ADD CONSTRAINT ck_m07_rx_amount CHECK (total_amount >= 0);

-- 已签名必须有签名信息
ALTER TABLE m07_prescription
    ADD CONSTRAINT ck_m07_rx_signed
    CHECK (status = 'DRAFT'
           OR (signed_at IS NOT NULL
               AND signer_name IS NOT NULL
               AND signature_hash IS NOT NULL));

-- 审核通过必须有审核人
ALTER TABLE m07_prescription
    ADD CONSTRAINT ck_m07_rx_reviewed
    CHECK (status NOT IN ('REVIEWED', 'DISPENSED')
           OR (reviewed_at IS NOT NULL AND reviewer_name IS NOT NULL));

-- 已退回必须有原因
ALTER TABLE m07_prescription
    ADD CONSTRAINT ck_m07_rx_returned
    CHECK (status <> 'RETURNED' OR return_reason IS NOT NULL);

-- 已发药必须有发药信息
ALTER TABLE m07_prescription
    ADD CONSTRAINT ck_m07_rx_dispensed
    CHECK (status <> 'DISPENSED'
           OR (dispensed_at IS NOT NULL AND dispense_no IS NOT NULL));

-- 已作废必须有原因
ALTER TABLE m07_prescription
    ADD CONSTRAINT ck_m07_rx_cancelled
    CHECK (status <> 'CANCELLED' OR cancel_reason IS NOT NULL);

-- 含抗菌药必须有级别
ALTER TABLE m07_prescription
    ADD CONSTRAINT ck_m07_rx_abx
    CHECK (has_antibiotic = FALSE OR antibiotic_level IS NOT NULL);

-- 待审核队列
CREATE INDEX idx_m07_rx_pending
    ON m07_prescription (created_at)
    WHERE status = 'SIGNED' AND is_deleted = FALSE;

-- 抗菌药统计
CREATE INDEX idx_m07_rx_abx
    ON m07_prescription (dept_id, doctor_id, created_at)
    WHERE has_antibiotic = TRUE AND is_deleted = FALSE;
```

**状态机**

```

        ┌──────────┐
        │  SIGNED  │  已签名（待审核）
```

```python
        "QOD": "隔日一次", "PRN": "必要时", "ONCE": "单次",
    }
    # 每日给药次数
    FREQ_PER_DAY = {
        "QD": 1, "BID": 2, "TID": 3, "QID": 4, "Q6H": 4,
        "Q4H": 6, "Q2H": 12, "Q1H": 24, "QN": 1, "QOD": 0.5,
        "PRN": 0, "ONCE": 1,
    }
    EYE_LABELS = {"OD": "右眼", "OS": "左眼", "OU": "双眼", "NA": ""}
    OCULAR_ROUTES = (
        "TOPICAL_EYE", "TOPICAL_EYE_OINTMENT", "SUBCONJUNCTIVAL",
        "RETROBULBAR", "INTRAVITREAL", "INTRAOCULAR_IRRIGATION",
    )
    LEVEL_LABELS = {
        "NON_RESTRICTED": "非限制使用级",
        "RESTRICTED": "限制使用级",
        "SPECIAL": "特殊使用级",
    }
    DROPS_PER_UNIT = 100

    def route_label(self) -> str:
        return self.ROUTE_LABELS.get(self.route, self.route)

    def freq_label(self) -> str:
        return self.FREQ_LABELS.get(self.frequency, self.frequency)

    def eye_label(self) -> str:
        return self.EYE_LABELS.get(self.eye_side, "")

    def level_label(self) -> str:
        return self.LEVEL_LABELS.get(self.antibiotic_level or "", "")

    def is_ocular(self) -> bool:
        return self.route in self.OCULAR_ROUTES

    def display_text(self) -> str:
        """药品显示文本"""
        name = self.drug_name or self.drug_code
        if self.spec:
            return f"{name} {self.spec}"
        return name

    def quantity_text(self) -> str:
        qty = float(self.quantity or 0)
        s = f"{qty:.0f}" if qty == int(qty) else f"{qty:g}"
        return f"{s}{self.unit or ''}"

    def build_usage_text(self) -> str:
        """生成规范用法文本"""
        parts = [self.route_label()]

        eye = self.eye_label()
        if eye:
            parts.append(eye)

        if self.single_dose:
            dose = float(self.single_dose)
            ds = f"{dose:.0f}" if dose == int(dose) else f"{dose:g}"
            parts.append(f"每次 {ds}{self.dose_unit or ''}")

        if not self.is_tapered:
            freq = self.freq_label()
            if freq:
                parts.append(freq)

        if self.duration_days:
            parts.append(f"共 {self.duration_days} 天")

        return "，".join(parts)

    def taper_text(self) -> str:
        """递减用药文本"""
        if not self.is_tapered or not self.taper_phases:
            return ""
        segs = []
        for p in self.taper_phases:
            freq = self.FREQ_LABELS.get(p.get("frequency", ""), "")
            segs.append(f"{freq}×{p.get('days', 0)}天")
        return "→".join(segs)

    def required_drops(self) -> int:
        """疗程所需滴数"""
        per_day = self.FREQ_PER_DAY.get(self.frequency, 0)
        if per_day <= 0:
            return 0
        days = self.duration_days or 0
        dose = float(self.single_dose or 1)
        return int(per_day * days * dose)

    def available_drops(self) -> int:
        """现有数量换算滴数"""
        return int(float(self.quantity or 0) * self.DROPS_PER_UNIT)

    def is_quantity_sufficient(self) -> bool:
        """数量是否足够覆盖疗程"""
        if not self.is_ocular():
            return True
        req = self.required_drops()
        if req == 0:
            return True
        return self.available_drops() >= req

    def suggested_quantity(self) -> float:
        """建议数量"""
        req = self.required_drops()
        if req == 0:
            return float(self.quantity or 1)
        return round(req / self.DROPS_PER_UNIT + 0.499, 0)

    def needs_eye_side(self) -> bool:
        return self.is_ocular()

    )

    FREQ_LABELS = M07PrescriptionItem.FREQ_LABELS

    def route_label(self) -> str:
        return self.ROUTE_LABELS.get(self.route, self.route)

    def preview_usage(self, eye_side: str = "NA") -> str:
        """预览用法（应用前）"""
        parts = [self.route_label()]
        eye = M07PrescriptionItem.EYE_LABELS.get(eye_side, "")
        if eye:
            parts.append(eye)
        parts.append(f"每次 {float(self.single_dose):g}"
                      f"{self.dose_unit or ''}")
        if not self.is_tapered:
            parts.append(self.FREQ_LABELS.get(self.frequency, ""))
        if self.duration_days:
            parts.append(f"共 {self.duration_days} 天")
        return "，".join(p for p in parts if p)
```
