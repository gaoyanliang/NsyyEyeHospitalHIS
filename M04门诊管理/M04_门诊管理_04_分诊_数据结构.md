**设计说明：**

- `m04_triage` 与 `m04_registration` 是 **1:1**（一个挂号只分诊一次，改分诊是更新同一条）
- `m04_queue_entry` 是队列快照，**每诊室每日一条记录一个患者**；改分诊时旧条目 `CANCELLED`，新建条目
- `queue_entry.position` 是**物化位置**，实际排序仍按 `(priority_weight DESC, triaged_at ASC)`
- `m04_triage_room` 是诊室的运行时状态表（是否启用、当前医生）
- `m04_triage_critical_handle` 记录危重患者的处置措施与时效

```sql
    ADD CONSTRAINT ck_m04_triage_color

    ON m04_triage (registration_id)

ALTER TABLE m04_triage
    ADD CONSTRAINT ck_m04_triage_triaged
    CHECK (status = 'PENDING'
           OR (triage_level IS NOT NULL AND triaged_at IS NOT NULL));

ALTER TABLE m04_triage
    ADD CONSTRAINT ck_m04_triage_reason
```

```sql
ALTER TABLE m04_queue_entry
    ADD CONSTRAINT ck_m04_queue_status

    ADD CONSTRAINT ck_m04_queue_weight
    CHECK (priority_weight >= 0 AND bonus_weight >= 0);

ALTER TABLE m04_queue_entry
    ADD CONSTRAINT ck_m04_queue_position
    CHECK (position IS NULL OR position > 0);

-- 一个分诊在一次活跃队列中只出现一次

    ON m04_queue_entry (triage_id)
    WHERE status IN ('WAITING', 'CALLED', 'IN_VISIT') AND is_deleted = FALSE;
```

**队列排序 SQL（关键）**

```sql
-- 取诊室候诊队列（按优先级 + 分诊时间）
SELECT id, queue_no, patient_name, triage_level, priority_weight
FROM m04_queue_entry
WHERE room_id = $1
  AND queue_date = CURRENT_DATE
  AND status = 'WAITING'
ORDER BY priority_weight DESC, triaged_at ASC
LIMIT 50;
```

> **为什么按 `triaged_at` 而非 `created_at`：** 分诊时间是患者实际等待起点，更公平。

```python
    TYPE_LABELS = {
        "SYMPTOM": "症状", "DIAGNOSIS": "诊断", "VITAL": "生命体征",
        "AGE": "年龄", "REG_TYPE": "挂号类型", "SPECIAL": "特殊标记",
    }

    def type_label(self) -> str:
        return self.TYPE_LABELS.get(self.rule_type, self.rule_type)

    def get_keywords(self) -> List[str]:
        if not self.keywords:
            return []
        return [k.strip() for k in self.keywords.split(",") if k.strip()]

    def get_symptom_tags(self) -> List[str]:
        if not self.symptom_tags:
            return []
        return [t.strip() for t in self.symptom_tags.split(",") if t.strip()]

    def matches_context(self, chief_complaint: str, tags: set,
                          vitals: dict, patient_age: Optional[int],
                          reg_type: Optional[str]) -> bool:
        """判断规则是否命中"""
        if self.rule_type == "SYMPTOM":
            for kw in self.get_keywords():
                if kw and kw in chief_complaint:
                    return True
            required_tags = set(self.get_symptom_tags())
            if required_tags and required_tags.issubset(tags):
                return True

        elif self.rule_type == "VITAL":
            cond = self.vital_condition or {}
            logic = cond.get("logic", "OR")
            results = []
            for key, rule in cond.items():
                if key == "logic" or not isinstance(rule, dict):
                    continue
                v = vitals.get(key)
                if v is None:
                    results.append(False)
                    continue
                ok = True
                if "gt" in rule and not float(v) > float(rule["gt"]):
                    ok = False
                if "gte" in rule and not float(v) >= float(rule["gte"]):
                    ok = False
                if "lt" in rule and not float(v) < float(rule["lt"]):
                    ok = False
                if "lte" in rule and not float(v) <= float(rule["lte"]):
                    ok = False
                results.append(ok)

            if logic == "AND":
                return bool(results) and all(results)
            return any(results)

        elif self.rule_type == "AGE":
            if patient_age is None:
                return False
            if self.age_min is not None and patient_age < self.age_min:
                return False
            if self.age_max is not None and patient_age > self.age_max:
                return False
            return True

        elif self.rule_type == "REG_TYPE":
            if not reg_type:
                return False
            types = [t.strip() for t in (self.reg_types or "").split(",") if t.strip()]
            return reg_type in types

        return False

    def get_matched_keywords(self, chief: str, tags: set) -> List[str]:
        matched = []
        for kw in self.get_keywords():
            if kw and kw in chief:
                matched.append(kw)
        for t in self.get_symptom_tags():
            if t in tags:
                matched.append(t)
        return matched

    def get_matched_value(self, vitals: dict) -> Optional[str]:
        cond = self.vital_condition or {}
        parts = []
        for key, rule in cond.items():
            if key == "logic" or not isinstance(rule, dict):
                continue
            v = vitals.get(key)
            if v is not None:
                parts.append(f"{key}={v}")
        return "; ".join(parts) if parts else None
```

```python
class TriageRuleBase(BaseModel):
    rule_name: str = Field(..., min_length=1, max_length=128)
    rule_type: Literal["SYMPTOM", "DIAGNOSIS", "VITAL",
                        "AGE", "REG_TYPE", "SPECIAL"]
    keywords: Optional[List[str]] = None
    symptom_tags: Optional[List[str]] = None
    vital_condition: Optional[dict] = None
    diag_keywords: Optional[List[str]] = None
    age_min: Optional[int] = Field(None, ge=0, le=150)
    age_max: Optional[int] = Field(None, ge=0, le=150)
    reg_types: Optional[List[str]] = None
    suggested_level: TRIAGE_LEVEL
    is_override: bool = False
    priority: int = 100
    is_enabled: bool = True
    description: Optional[str] = Field(None, max_length=500)

    @model_validator(mode="after")
    def validate_rule(self):
        if self.rule_type == "SYMPTOM" and not (self.keywords or self.symptom_tags):
            raise ValueError("症状规则必须指定关键词或症状标签")
        if self.rule_type == "VITAL" and not self.vital_condition:
            raise ValueError("生命体征规则必须指定条件")
        if self.rule_type == "AGE" and self.age_min is None and self.age_max is None:
            raise ValueError("年龄规则必须指定年龄范围")
        if self.age_min is not None and self.age_max is not None:
            if self.age_max < self.age_min:
                raise ValueError("年龄上限不可小于下限")
        return self
```
