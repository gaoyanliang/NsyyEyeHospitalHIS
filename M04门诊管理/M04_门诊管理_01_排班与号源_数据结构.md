## 4. 数据结构设计

### 4.1 ER 关系图

```
┌──────────────────────────────────────────────────────────────────────┐
│ M01 主数据（引用，不建物理外键）                                       │
│ mdm_employee（医生）/ mdm_department（科室）/ md_dict_item（号别）     │
└──────────────────────────────┬───────────────────────────────────────┘
                               │ doctor_id / dept_id
                               ▼
┌──────────────────────────────────┐
│ m04_schedule_template            │  周期规则
│  · template_code                 │
│  · doctor_id / weekday / period  │
│  · effective_from / effective_to │
└──────────────┬───────────────────┘
               │ 生成
               ▼
┌──────────────────────────────────┐
│ m04_schedule                     │  某日的班次
│  · schedule_date / period        │
│  · doctor_id / dept_id           │
│  · total_quota / used_quota      │
│  · status（NORMAL/SUSPENDED）    │
└──────────────┬───────────────────┘
               │
      ┌────────┴─────────┐
      ▼                  ▼
┌────────────────────┐  ┌──────────────────────────┐
│ m04_schedule_period│  │ m04_reg_slot             │  号源
│  时段               │◄─│  · slot_no               │
│  · period_text     │  │  · period_id             │
│  · quota           │  │  · status（AVAILABLE...）│
└────────────────────┘  │  · patient_id / biz_no   │
                        │  · source_type           │
                        └──────────────────────────┘

支撑表：
┌────────────────────────┐  ┌────────────────────────┐
│ m04_slot_reserve_rule  │  │ m04_schedule_change_   │
│ （留号规则，按科室/号别）│  │ log（变更日志，月分区） │
└────────────────────────┘  └────────────────────────┘
┌────────────────────────┐  ┌────────────────────────┐
│ m04_schedule_generate_ │  │ m04_holiday            │
│ task（生成任务）        │  │ （节假日）              │
└────────────────────────┘  └────────────────────────┘
┌────────────────────────┐  ┌────────────────────────┐
│ m04_schedule_config    │  │ m04_slot_reconcile_log │
│ （配置，单行）          │  │ （对账日志）            │
└────────────────────────┘  └────────────────────────┘
```

**设计说明：**

- `m04_schedule_template` 是**周期规则**，`m04_schedule` 是**实例**（某一天的班次）
- `m04_reg_slot` 是号源明细，**一行一个号**；号源状态是并发控制的核心
- **号源库存权威在 Redis**（`m04:slot:stock:{schedule_id}`），DB 是持久化
- `used_quota` 是冗余计数，与 `m04_reg_slot` 的 BOOKED 数一致，由对账任务保证
- 停诊不删除排班，`status = SUSPENDED`，保留可追溯

**索引设计**

| 索引名 | 类型 | 字段 | 说明 |
|---|---|---|---|
| pk_m04_schedule_template | PRIMARY | id | 主键 |
| uk_m04_tpl_code | UNIQUE | template_code | 编码唯一 |
| idx_m04_tpl_doctor | NORMAL | doctor_id, is_enabled | 按医生 |
| idx_m04_tpl_dept | NORMAL | dept_id, is_enabled | 按科室 |

**约束说明**

```sql
ALTER TABLE m04_schedule_template
    ADD CONSTRAINT ck_m04_tpl_code
    CHECK (template_code ~ '^TPL-[A-Za-z0-9-]{1,28}$');

ALTER TABLE m04_schedule_template
    ADD CONSTRAINT ck_m04_tpl_repeat
    CHECK (repeat_type IN ('WEEKLY', 'BIWEEKLY'));

-- 生效区间
ALTER TABLE m04_schedule_template
    ADD CONSTRAINT ck_m04_tpl_effective
    CHECK (effective_to IS NULL OR effective_to >= effective_from);

-- 号数上限
ALTER TABLE m04_schedule_template
    ADD CONSTRAINT ck_m04_tpl_quota
    CHECK (total_quota >= 0 AND total_quota <= 200);

-- 留号比例合计 ≤ 50%
ALTER TABLE m04_schedule_template
    ADD CONSTRAINT ck_m04_tpl_reserve
    CHECK (reserve_onsite_pct + reserve_emergency_pct + reserve_green_pct <= 50);

-- 时段参数
ALTER TABLE m04_schedule_template
    ADD CONSTRAINT ck_m04_tpl_period
    CHECK (period_minutes IN (15, 20, 30, 60) AND per_period_quota >= 0);

-- 防重复：同医生同日同时段只能有一个启用模板
CREATE UNIQUE INDEX uk_m04_tpl_doctor_slot
    ON m04_schedule_template (doctor_id, weekday, period)
    WHERE is_enabled = TRUE AND is_deleted = FALSE;
```

> **注意：** 最后一条唯一索引较严格（同一医生同一周的同一班次只能有一个模板）。若需支持"某周特殊排班"，需将 `repeat_type = BIWEEKLY` 纳入唯一键，或将模板改为多点叠加。本期**采用单一模板**策略。

**状态机**

```
      ┌──────────┐
      │  LOCKED  │  生成后未开放
      └────┬─────┘
           │ 开放
           ▼
      ┌──────────┐
```

**约束说明**

```sql
-- 关键：同一排班内号序唯一
CREATE UNIQUE INDEX uk_m04_slot_no
    ON m04_reg_slot (schedule_id, slot_no)
    WHERE is_deleted = FALSE;

ALTER TABLE m04_reg_slot
    ADD CONSTRAINT ck_m04_slot_status
    CHECK (status IN ('AVAILABLE', 'LOCKED', 'RESERVED',
                       'BOOKED', 'CHECKED_IN', 'CANCELLED', 'CLOSED'));

ALTER TABLE m04_reg_slot
    ADD CONSTRAINT ck_m04_slot_source
    CHECK (source_type IN ('OPEN', 'ONSITE', 'EMERGENCY',
                            'GREEN', 'ADDED', 'FOLLOWUP'));

ALTER TABLE m04_reg_slot
    ADD CONSTRAINT ck_m04_slot_biz_type
    CHECK (biz_type IS NULL OR biz_type IN (
        'APPOINTMENT', 'REGISTRATION', 'SPECIAL'));

-- LOCKED 必须有锁信息
ALTER TABLE m04_reg_slot
    ADD CONSTRAINT ck_m04_slot_lock
    CHECK (status <> 'LOCKED' OR (lock_id IS NOT NULL
                                   AND lock_expire_at IS NOT NULL));

-- BOOKED/CHECKED_IN 必须有患者
ALTER TABLE m04_reg_slot
    ADD CONSTRAINT ck_m04_slot_booked
    CHECK (status NOT IN ('BOOKED', 'CHECKED_IN')
```

```python
        """解析周几"""
        if not self.weekday:
            return []
        return [int(x) for x in self.weekday.split(",") if x.strip().isdigit()]

    def get_periods(self) -> list:
        """解析班次"""
        if not self.period:
            return []
        return [x.strip() for x in self.period.split(",") if x.strip()]

        """指定日期是否在生效区间"""
        if d < self.effective_from:
            return False
        if self.effective_to and d > self.effective_to:
            return False
        return True

    def matches(self, d: date) -> bool:
        """指定日期是否命中模板"""
        if not self.is_enabled or self.is_deleted:
            return False
        if not self.is_effective(d):
            return False
        if d.isoweekday() not in self.get_weekdays():
            return False

        # 双周模板
        if self.repeat_type == "BIWEEKLY" and self.repeat_anchor_date:
            weeks = (d - self.repeat_anchor_date).days // 7
            if weeks % 2 != 0:
                return False

        return True
```

```python
    template_name: Optional[str] = Field(None, max_length=128)
    dept_id: int
    doctor_id: int

    @field_validator("weekday")
    @classmethod
    def validate_weekday(cls, v):
        parts = [x.strip() for x in v.split(",") if x.strip()]
        if not parts:
            raise ValueError("周几不可为空")
        for p in parts:
            if not p.isdigit() or not (1 <= int(p) <= 7):
                raise ValueError(f"周几取值应在 1~7，收到 {p}")
        return ",".join(sorted(set(parts), key=int))

    @field_validator("period")
    @classmethod
    def validate_period(cls, v):
        allowed = {"AM", "PM", "NIGHT", "ALL_DAY"}
        parts = [x.strip() for x in v.split(",") if x.strip()]
        if not parts:
            raise ValueError("班次不可为空")
        for p in parts:
            if p not in allowed:
                raise ValueError(f"班次必须是 {allowed} 之一，收到 {p}")
        return ",".join(parts)

    @model_validator(mode="after")
    def validate_template(self):
        if self.effective_to and self.effective_to < self.effective_from:
            raise ValueError("生效结束日期不能早于开始日期")

        total_reserve = (self.reserve_onsite_pct + self.reserve_emergency_pct
                         + self.reserve_green_pct)
        if total_reserve > 50:
            raise ValueError(f"留号比例合计不可超过 50%，当前 {total_reserve}%")

        if self.period_minutes not in (15, 20, 30, 60):
            raise ValueError("时段时长必须是 15/20/30/60 分钟")

        # 总号数计算校验
        if self.period_split_enabled and self.total_quota == 0:
            raise ValueError("启用时段划分时总号数不可为 0")

        if self.repeat_type == "BIWEEKLY" and not self.repeat_anchor_date:
            raise ValueError("双周重复必须指定锚点日期")

        return self

class ScheduleConfigUpdate(BaseModel):
    period_split_enabled: bool = True
    period_minutes: int = Field(30, ge=15, le=60)
    per_period_quota: int = Field(4, ge=0, le=50)
    max_quota_per_schedule: int = Field(200, ge=1, le=500)
    max_add_per_schedule: int = Field(5, ge=1, le=50)
    max_add_per_day: int = Field(20, ge=1, le=100)

    reserve_onsite_pct: int = Field(20, ge=0, le=50)
    reserve_emergency_pct: int = Field(5, ge=0, le=50)
    reserve_green_pct: int = Field(5, ge=0, le=50)
    reserve_release_mode: Literal["NONE", "BEFORE_30MIN",
                                    "AT_START", "MANUAL"] = "BEFORE_30MIN"

    slot_lock_seconds: int = Field(300, ge=60, le=1800)
    onsite_advance_days: int = Field(0, ge=0, le=30)
    portal_advance_days: int = Field(7, ge=0, le=30)
    inhouse_advance_days: int = Field(30, ge=0, le=90)
    auto_generate_enabled: bool = False

    in_hospital_max_per_doctor: int = Field(30, ge=1, le=100)
    online_reg_enabled: bool = True

    version_no: int = Field(..., ge=1)

    @model_validator(mode="after")
    def validate_config(self):
        total = (self.reserve_onsite_pct + self.reserve_emergency_pct
                 + self.reserve_green_pct)
        if total > 50:
            raise ValueError(f"留号比例合计不可超过 50%，当前 {total}%")

        if self.portal_advance_days > self.inhouse_advance_days:
            raise ValueError("门户提前天数不应超过院内提前天数")

        return self
```
