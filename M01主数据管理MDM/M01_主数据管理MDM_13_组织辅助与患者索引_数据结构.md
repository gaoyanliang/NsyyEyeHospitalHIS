    def compute_status(self) -> str:
        """计算证书状态"""
        if self.cert_status == "REVOKED":
            return "REVOKED"
        if self.is_long_term or self.expire_date is None:
            return "VALID"
        from datetime import date as date_cls
        delta = (self.expire_date - date_cls.today()).days
        if delta < 0:
            return "EXPIRED"
        if delta <= 60:
            return "EXPIRING"
        return "VALID"

        if self.is_long_term or not self.expire_date:
            return None
        from datetime import date as date_cls
        return (self.expire_date - date_cls.today()).days

    cost_method: Optional[Literal["FIFO", "WEIGHTED_AVG", "SPECIFIC"]] = None
    is_central: bool = False
    parent_warehouse_id: Optional[int] = None
    address: Optional[str] = Field(None, max_length=255)
    phone: Optional[str] = Field(None, max_length=32)
    allow_negative: bool = False
    sort_no: int = 999
    remark: Optional[str] = Field(None, max_length=512)

    @model_validator(mode="after")
    def validate_warehouse(self):
        # 高值耗材柜强制成本核算 + 个别计价
        if self.warehouse_type == "HIGH_VALUE_CABINET":
            if not self.is_cost_account:
                raise ValueError("高值耗材柜必须参与成本核算")
            if self.cost_method != "SPECIFIC":
                raise ValueError("高值耗材柜必须使用「个别计价」成本方法")
        # 零售仓必须成本核算
        if self.is_retail and not self.is_cost_account:
            raise ValueError("零售仓必须参与成本核算")
        # 参与成本核算必须有方法
        if self.is_cost_account and not self.cost_method:
            raise ValueError("参与成本核算时必须指定成本核算方法")
        # 二级库必须有上级
        if self.warehouse_type == "OR_SUB_STORE" and not self.parent_warehouse_id:
            raise ValueError("手术室二级库必须指定上级仓库")
        return self

class StaffCertBase(BaseModel):
    staff_id: int
    cert_type: Literal[
        "PHYSICIAN_QUALIFY", "PHYSICIAN_PRACTICE", "NURSE_PRACTICE",
        "TITLE_CERT", "LARGE_EQUIPMENT", "OPTOMETRIST",
        "ORTHO_K_FIT", "REFRACTIVE_TRAINING", "INFECTION_TRAINING", "OTHER",
    ]
    cert_no: str = Field(..., min_length=1, max_length=64)
    issuing_authority: Optional[str] = Field(None, max_length=128)
    issue_date: Optional[date] = None
    expire_date: Optional[date] = None
    is_long_term: bool = False
    practice_scope: Optional[str] = Field(None, max_length=255)
    practice_location: Optional[str] = Field(None, max_length=128)
    attachment_ids: Optional[List[int]] = None
    is_required_cert: bool = False
    remark: Optional[str] = Field(None, max_length=512)

    @model_validator(mode="after")
    def validate_cert(self):
        # 长期有效则不填到期日
        if self.is_long_term and self.expire_date:
            raise ValueError("长期有效证书不应填写有效期")
        # 非长期必须有到期日
        if not self.is_long_term and not self.expire_date:
            raise ValueError("非长期有效证书必须填写有效期")
        # 到期日必须晚于发证日
        if self.issue_date and self.expire_date:
            if self.expire_date <= self.issue_date:
                raise ValueError("有效期必须晚于发证日期")
        # 发证日期不能晚于今天
        if self.issue_date and self.issue_date > date.today():
            raise ValueError("发证日期不能晚于今天")
        # 必需证书必须有附件
        if self.is_required_cert and not self.attachment_ids:
            raise ValueError("必需证书必须上传扫描件")
        return self
