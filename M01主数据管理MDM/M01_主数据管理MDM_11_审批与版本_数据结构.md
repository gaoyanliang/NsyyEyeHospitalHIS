class ApprovalChainRule(Base, TimestampMixin, SoftDeleteMixin):
    """审批链规则"""
    __tablename__ = "md_approval_chain_rule"

    id: Mapped[int] = mapped_column(BigInteger, primary_key=True, autoincrement=True)
    rule_code: Mapped[str] = mapped_column(String(32), nullable=False)
    rule_name: Mapped[str] = mapped_column(String(128), nullable=False)
    domain_code: Mapped[str] = mapped_column(String(16), nullable=False)
    change_types: Mapped[str] = mapped_column(String(255), nullable=False)
    match_condition: Mapped[Optional[dict]] = mapped_column(JSONB)
    priority: Mapped[int] = mapped_column(Integer, default=100, nullable=False)
    sla_hours: Mapped[int] = mapped_column(Integer, default=24, nullable=False)
    node_sla_hours: Mapped[int] = mapped_column(Integer, default=24, nullable=False)
    allow_batch_approve: Mapped[bool] = mapped_column(Boolean, default=True, nullable=False)
    allow_self_approve: Mapped[bool] = mapped_column(Boolean, default=False, nullable=False)
    require_signature: Mapped[bool] = mapped_column(Boolean, default=True, nullable=False)
    is_default: Mapped[bool] = mapped_column(Boolean, default=False, nullable=False)
    status: Mapped[str] = mapped_column(String(16), default="ENABLED", nullable=False)
    description: Mapped[Optional[str]] = mapped_column(String(500))
    created_by: Mapped[Optional[int]] = mapped_column(BigInteger)
    updated_by: Mapped[Optional[int]] = mapped_column(BigInteger)

    nodes: Mapped[List["ApprovalChainNode"]] = relationship(
        back_populates="rule", lazy="selectin",
        order_by="ApprovalChainNode.node_seq")

    __table_args__ = (
        UniqueConstraint("rule_code", "is_deleted", name="uk_chain_rule_code"),
        Index("idx_chain_rule_domain", "domain_code", "status", "priority"),
        {"comment": "审批链规则"},
    )

    def match(self, ctx: dict) -> bool:
        """判断上下文是否命中该规则"""
        if self.status != "ENABLED":
            return False
        if ctx.get("change_type") not in (self.change_types or "").split(","):
            return False
        cond = self.match_condition or {}
        if not cond:
            return True

        for key, expected in cond.items():
            actual = ctx.get(key)
            if isinstance(expected, list):
                if actual not in expected:
                    return False
            elif isinstance(expected, dict):
                # 阈值型条件
                field = expected.get("field")
                op = expected.get("operator")
                val = expected.get("value")
                actual_val = ctx.get("data", {}).get(field) if field else actual
                if not self._compare(actual_val, op, val):
                    return False
            else:
                if actual != expected:
                    return False
        return True

    @staticmethod
    def _compare(actual, op: str, expected) -> bool:
        if actual is None:
            return False
        try:
            a, e = float(actual), float(expected)
        except (TypeError, ValueError):
            return actual == expected
        return {
            "GT": a > e, "GTE": a >= e, "LT": a < e,
            "LTE": a <= e, "EQ": a == e, "NE": a != e,
        }.get(op, False)

from datetime import datetime
from decimal import Decimal
from typing import Optional, List, Literal, Any

from pydantic import BaseModel, Field, ConfigDict, field_validator, model_validator


CHANGE_TYPES = Literal[
    "CREATE", "UPDATE", "DISABLE", "DELETE",
    "IMPORT", "PRICE_ADJUST", "ROLLBACK",
]
URGENCY = Literal["NORMAL", "URGENT", "CRITICAL"]
REQ_STATUS = Literal[
    "DRAFT", "PENDING", "APPROVING", "APPROVED",
    "REJECTED", "WITHDRAWN", "EFFECTIVE", "FAILED",
]


# ==================== 变更单 ====================
class ChangeRequestCreate(BaseModel):
    change_type: CHANGE_TYPES
    domain_code: Literal["ORG", "PAT", "MAT", "CLI", "FEE", "AUTH", "DICT", "OPT"]
    object_type: str = Field(..., max_length=64)
    object_id: Optional[int] = None
    object_code: Optional[str] = Field(None, max_length=64)
    object_name: Optional[str] = Field(None, max_length=255)
    before_json: Optional[dict] = None
    after_json: Optional[dict] = None
    reason: str = Field(..., min_length=5, max_length=500)
    urgency: URGENCY = "NORMAL"
    urgency_reason: Optional[str] = Field(None, max_length=500)
    attachment_ids: Optional[List[int]] = None
    apply_dept_id: Optional[int] = None
    source: Literal["PC", "MOBILE", "API", "IMPORT"] = "PC"
    remark: Optional[str] = Field(None, max_length=512)

    @model_validator(mode="after")
    def validate_request(self):
        # 更新/停用/删除必须有对象
        if self.change_type in ("UPDATE", "DISABLE", "DELETE") and not self.object_id:
            raise ValueError(f"{self.change_type} 类型必须指定 object_id")
        # 更新必须有变更前快照
        if self.change_type == "UPDATE" and not self.before_json:
            raise ValueError("修改类型必须提供变更前快照 before_json")
        # 新增不需要 before_json
        if self.change_type == "CREATE" and self.before_json:
            raise ValueError("新增类型不应提供 before_json")
        # 必须有变更后数据
        if self.change_type in ("CREATE", "UPDATE") and not self.after_json:
            raise ValueError("必须提供变更后数据 after_json")
        # 紧急必须有理由
        if self.urgency in ("URGENT", "CRITICAL") and not self.urgency_reason:
            raise ValueError("紧急变更必须填写紧急理由")
        return self

class ApproveAction(BaseModel):
    """审批动作"""
    action: Literal["APPROVE", "REJECT", "TRANSFER", "ADD_SIGN"]
    opinion: Optional[str] = Field(None, max_length=500)
    signature_id: Optional[int] = None
    transfer_to: Optional[int] = Field(None, description="转办目标人（TRANSFER 必填）")
    transfer_reason: Optional[str] = Field(None, max_length=255)
    add_sign_to: Optional[int] = Field(None, description="加签目标人（ADD_SIGN 必填）")
    add_sign_position: Optional[Literal["BEFORE", "AFTER"]] = "AFTER"
    version_no: int = Field(..., ge=1)

    @model_validator(mode="after")
    def validate_action(self):
        if self.action == "REJECT" and not self.opinion:
            raise ValueError("驳回必须填写审批意见")
        if self.action == "TRANSFER":
            if not self.transfer_to:
                raise ValueError("转办必须指定转办人")
            if not self.transfer_reason:
                raise ValueError("转办必须填写理由")
        if self.action == "ADD_SIGN" and not self.add_sign_to:
            raise ValueError("加签必须指定加签人")
        return self

    node_seq: int = Field(..., ge=1)
    node_name: str = Field(..., max_length=64)

    ]

class ChainRuleCreate(ChainRuleBase):
    rule_code: str = Field(..., pattern=r"^ACR[A-Z0-9_]{2,28}$")
    nodes: List[ChainNodeSchema] = Field(..., min_length=1)

    @model_validator(mode="after")
    def validate_nodes(self):
        seqs = [n.node_seq for n in self.nodes]
        if len(seqs) != len(set(seqs)):
            raise ValueError("节点序号重复")
        if sorted(seqs) != list(range(1, len(seqs) + 1)):
            raise ValueError("节点序号必须从 1 开始连续")
        approve_nodes = [n for n in self.nodes if n.node_type == "APPROVE"]
        if not approve_nodes:
            raise ValueError("至少需要一个审批节点")
        return self
